C++ 2D Arrays Game Logic Two Players 
  1#include <iostream>
  2#include <limits>
  3using namespace std;
  4
  5// ---------- Game board ----------
  6char board[3][3] = {
  7    {'1', '2', '3'},
  8    {'4', '5', '6'},
  9    {'7', '8', '9'}
 10};
 11
 12char currentPlayer = 'X';
 13
 14// ---------- Display the board ----------
 15void displayBoard() {
 16    cout << "\n";
 17    for (int i = 0; i < 3; ++i) {
 18        cout << " " << board[i][0] << " | " << board[i][1] << " | " << board[i][2] << "\n";
 19        if (i < 2) cout << "---+---+---\n";
 20    }
 21    cout << "\n";
 22}
 23
 24// ---------- Check for a winner ----------
 25// Returns 'X' or 'O' if someone won, 'D' for draw, or ' ' if game continues.
 26char checkResult() {
 27    // Rows
 28    for (int i = 0; i < 3; ++i)
 29        if (board[i][0] == board[i][1] && board[i][1] == board[i][2])
 30            return board[i][0];
 31
 32    // Columns
 33    for (int j = 0; j < 3; ++j)
 34        if (board[0][j] == board[1][j] && board[1][j] == board[2][j])
 35            return board[0][j];
 36
 37    // Diagonals
 38    if (board[0][0] == board[1][1] && board[1][1] == board[2][2])
 39        return board[0][0];
 40    if (board[0][2] == board[1][1] && board[1][1] == board[2][0])
 41        return board[0][2];
 42
 43    // Check for empty cells (game still going)
 44    for (int i = 0; i < 3; ++i)
 45        for (int j = 0; j < 3; ++j)
 46            if (board[i][j] != 'X' && board[i][j] != 'O')
 47                return ' ';   // game continues
 48
 49    return 'D';  // draw
 50}
 51
 52// ---------- Place a mark ----------
 53bool placeMove(int cell) {
 54    if (cell < 1 || cell > 9) return false;
 55
 56    int row = (cell - 1) / 3;
 57    int col = (cell - 1) % 3;
 58
 59    if (board[row][col] == 'X' || board[row][col] == 'O')
 60        return false;          // cell already taken
 61
 62    board[row][col] = currentPlayer;
 63    return true;
 64}
 65
 66// ---------- Main game loop ----------
 67int main() {
 68    cout << "=== Tic-Tac-Toe ===\n";
 69    cout << "Player 1 = X   |   Player 2 = O\n";
 70    cout << "Enter a number (1-9) to place your mark.\n";
 71
 72    char result = ' ';
 73
 74    while (result == ' ') {
 75        displayBoard();
 76        cout << "Player " << currentPlayer << ", enter your move: ";
 77
 78        int move;
 79        if (!(cin >> move)) {
 80            cin.clear();
 81            cin.ignore(numeric_limits<streamsize>::max(), '\n');
 82            cout << "Invalid input. Please enter a number between 1 and 9.\n";
 83            continue;
 84        }
 85
 86        if (!placeMove(move)) {
 87            cout << "Invalid move! Try again.\n";
 88            continue;
 89        }
 90
 91        result = checkResult();
 92
 93        if (result == ' ')
 94            currentPlayer = (currentPlayer == 'X') ? 'O' : 'X';
 95    }
 96
 97    displayBoard();
 98
 99    if (result == 'D')
100        cout << "It's a draw!\n";
101    else
102        cout << "Player " << result << " wins! Congratulations!\n";
103
104    return 0;
105}

    std::getline(ss, email, '|');

    }

// ── File persistence ─────────────────────────────────────────

void ContactManager::loadContacts() {
    contacts.clear();
    std::ifstream fin(filename);
    if (!fin.is_open()) return;

    Contact c;
    while (c.loadFromFile(fin))
        contacts.push_back(c);

    fin.close();
    generateNextId();
}

void ContactManager::saveContacts() const {
    std::ofstream fout(filename, std::ios::trunc);
    if (!fout.is_open()) {
        std::cerr << "\n  [!] Error: Could not open file for writing.\n";
        return;
    }
    for (const auto& c : contacts)
        c.saveToFile(fout);
    fout.close();
}

// ── Helpers ──────────────────────────────────────────────────

int ContactManager::findIndexById(int id) const {
    for (size_t i = 0; i < contacts.size(); ++i)
        if (contacts[i].getId() == id) return static_cast<int>(i);
    return -1;
}

void ContactManager::generateNextId() {
    nextId = 1;
    for (const auto& c : contacts)
        if (c.getId() >= nextId) nextId = c.getId() + 1;
}
